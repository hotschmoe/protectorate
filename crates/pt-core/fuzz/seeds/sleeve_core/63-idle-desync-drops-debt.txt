#scenario c662b56659f569df
# A debt from a crashed turn, after the resume reached Ready and before the
# continuation is sent, is dropped when that idle incarnation desyncs.
# The resume warns once and does not continue. step 2 is the kill, ahead of
# the continuation still waiting to be sent.
spawn debt /w/debt
run
obs debt ready
send debt m1
run
expect debt activity=Working
exit debt
run
obs debt ready
obs debt desync
step 2
expect debt g=3 activity=Booting life=Launching guard=Resuming
expect-notice TurnNeedsPrompt count=1
expect-effect continue:* count=0
run
obs debt ready
run
expect debt activity=Idle life=Running guard=Armed
expect-effect continue:* count=0
expect-notice TurnNeedsPrompt count=1
