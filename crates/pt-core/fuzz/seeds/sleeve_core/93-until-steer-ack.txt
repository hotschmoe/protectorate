#scenario ff1d9004dc164598
# A refused steer cannot arm a loop; a confirmed steer can.
spawn root /w/a harness=codex
run
obs root ready
send root m1
run
policy prompt=nack
send root m2 steer until=3
run
expect root until_rounds=none msg:m2=Rejected
obs root continuing
run
expect-effect until:* count=0
policy prompt=ack
send root m3
run
send root m4 steer until=2
run
expect root until_rounds=0 msg:m4=Delivered
obs root continuing
run
expect root until_rounds=1
expect-effect until:* state=Confirmed attempt=1 count=1
