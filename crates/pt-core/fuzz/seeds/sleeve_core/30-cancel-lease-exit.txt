# including across recovery. After the lease, the node's own answer decides the outcome.

crash

# Before the lease, any answer that ends a cancelled job ends it as Cancelled and acknowledges
# collection: journals replay this path, so it stays as it was.
job early-exit
policy poll=0
run
cancel-job 1
job-report 1 exited=2
expect-job 1 state=Cancelled
expect-effect cancel:1:1:* state=Confirmed count=1
expect-effect collected:1:1 state=Intent count=1
expect-notice JobFinished count=2
run
