#scenario d776614d301efa8f
# A second crash of the same owner turn, after the resume window, resumes again
# and does not send a second continuation.
spawn root /w/a
run
obs root ready
send root m1
run
expect root activity=Working
exit root
run
obs root ready
run
expect root activity=Working
expect-effect continue:0:1 state=Confirmed count=1
tick 11m
exit root
run
obs root ready
run
expect root activity=Idle
expect-effect continue:0:1 state=Confirmed count=1
expect-effect continue:* count=1
expect-notice TurnNeedsPrompt count=1
