#scenario 462d8f67c853c15e
# One continuation after a native resume of an interrupted turn. A second journaled Continue
# does nothing, and a crash of the continued turn warns instead of sending another. None
# after a planned stop, a hub restart of an idle sleeve, or a manual resleeve. A pane warns
# instead. A rejected prompt warns once. A restart before the prompt is journaled keeps the
# debt, and the later generation sends it.
spawn root /w/a
run
obs root ready
send root m1
run
expect root activity=Working
exit root
run
obs root ready
run
expect root activity=Working
expect-notice TurnNeedsPrompt count=0
expect-effect continue:0:1 state=Confirmed count=1
expect root msg:continue:0:1=Delivered
continue root
expect-effect continue:* count=1
exit root
run
obs root ready
run
expect root activity=Idle
expect-effect continue:* count=1
expect-effect continue:0:2 count=0
expect-notice TurnNeedsPrompt count=1
stop root
run
expect root life=Stopped

expect owed activity=Working
exit owed

crash
run

run
expect owed activity=Working
expect-effect continue:1:1 state=Confirmed count=1
expect-effect continue:1:2 count=0
stop owed
run
expect owed life=Stopped
spawn idle /w/b
run
obs idle ready
run

run
