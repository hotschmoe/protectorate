#scenario 5833b641c74900e7
# One continuation after a native resume of an interrupted turn. A second journaled Continue
# does nothing, and the next interruption owes one more. None after a planned stop, a hub
# restart of an idle sleeve, or a manual resleeve. A pane warns instead. A rejected prompt
# warns once. A restart before the prompt is journaled retargets the debt to the new generation.
spawn root /w/a
run
obs root ready
send root m1
run
expect root activity=Working
exit root
run
obs root ready
run
expect root activity=Working
expect-notice TurnNeedsPrompt count=0
expect-effect continue:0:2 state=Confirmed count=1
expect root msg:continue:0:2=Delivered
continue root
expect-effect continue:* count=1
exit root
run
obs root ready
run
expect root activity=Working
expect-effect continue:* count=2
expect-effect continue:0:3 state=Confirmed count=1
stop root
run
expect root life=Stopped
