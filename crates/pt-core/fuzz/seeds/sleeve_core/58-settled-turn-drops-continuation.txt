# continuation. The journal has that settle, so a later hub start does not warn.

run

run
obs settled ready
obs settled settled
expect settled activity=Idle
expect-effect continue:* count=0
expect-notice TurnNeedsPrompt count=0
commit
crash
expect-effect continue:* count=0
