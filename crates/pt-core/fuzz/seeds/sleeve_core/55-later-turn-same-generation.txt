#scenario 6a2f1f5fe8c143bb
# A later owner message in a generation that already continued is its own turn.
# A crash of that turn gets one continuation.
spawn root /w/a
run
obs root ready
send root m1
run
expect root activity=Working
exit root
run
obs root ready
run
expect root activity=Working
expect-effect continue:0:1 state=Confirmed count=1
obs root settled
expect root activity=Idle
send root m2
run
expect root activity=Working
exit root
run
obs root ready
run
expect root activity=Working
expect-effect continue:0:2 state=Confirmed count=1
expect-effect continue:* count=2
expect-notice TurnNeedsPrompt count=0
