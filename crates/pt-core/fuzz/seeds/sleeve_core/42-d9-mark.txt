#scenario 88a96f1b97494428
# A D9 send marks the recipient and the children it already has. A child spawned later inherits.
# A marked spawn marks that sleeve. A send from a marked sleeve marks its recipient, so a relay
# cannot leave the parent unmarked. An ordinary send from an unmarked sleeve does not spread it.
# Envoy is never marked. The mark is part of the event: a crash keeps it and replay rebuilds it.
spawn envoy /w/e harness=codex
spawn root /w/a
spawn leaf /wt/leaf parent=root
spawn kid /wt/kid parent=root
spawn plain /w/b
spawn plainchild /wt/pc parent=plain
spawn relay /w/d
spawn hop /wt/hop parent=relay d9
run
obs envoy ready
obs root ready
obs leaf ready
obs kid ready
obs plain ready
obs plainchild ready
obs relay ready
obs hop ready
send root m1 d9 from=envoy
expect root d9=true
expect leaf d9=true
expect kid d9=true
expect envoy d9=false
spawn later /wt/later parent=root
expect later d9=true
spawn marked /w/c d9
expect marked d9=true
spawn child /wt/child parent=marked
expect child d9=true
send plainchild m2 from=plain
expect plain d9=false
expect plainchild d9=false
expect relay d9=false
expect hop d9=true
send relay hop1 from=hop
expect relay d9=true
expect hop d9=true
send envoy m3 d9 from=root
expect envoy d9=false
commit
crash empty
expect root d9=true
expect leaf d9=true
expect kid d9=true
expect later d9=true
expect marked d9=true
expect child d9=true
expect plain d9=false
expect plainchild d9=false
expect relay d9=true
expect hop d9=true
expect envoy d9=false
